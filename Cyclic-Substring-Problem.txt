import java.util.*;

public class CyclicSum {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.next();
        int n = s.length();
        int maxSum = 0;

        // We only need to check starting positions up to n
        for (int i = 0; i < n; i++) {
            boolean[] visited = new boolean[26];
            int currentSum = 0;
            
            // A unique substring can at most be 26 characters long
            for (int j = 0; j < Math.min(n, 26); j++) {
                char c = s.charAt((i + j) % n); // Use modulo for cyclicity
                if (visited[c - 'a']) break;
                
                visited[c - 'a'] = true;
                currentSum += (c - 'a' + 1);
                maxSum = Math.max(maxSum, currentSum);
            }
        }
        System.out.println(maxSum);
    }
}
